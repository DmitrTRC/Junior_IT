1Коробочки — помним?

let imya = "Саня"let ochki = 0console.log("Привет, " + imya + "! Очков: " + ochki)Привет, Саня! Очков: 0// ловушка после каникул:console.log("7" + 1)71
Не восемь: к тексту плюс приклеивает, а не прибавляет.

2Спросить у человека

let otvet = Number(prompt("Сколько будет 7 × 3?"))console.log(otvet)21
Без Number в коробочке был бы текст "21" — и дальше всё сломается.

3Проверить ответ

let otvet = Number(prompt("Сколько будет 7 × 3?"))if (otvet === 21) {  console.log("✅ Верно!")} else {  console.log("❌ Мимо. Правильно: 21")}
Показать на секунду if (otvet = 21) — срабатывает всегда — и стереть.

4Одна попытка целиком

let chislo = 7let i = 3let otvet = Number(prompt(chislo + " × " + i + " = ?"))if (otvet === chislo * i) {  console.log("✅ Верно!")} else {  console.log("❌ Мимо. Правильно: " + (chislo * i))}
Число 21 нигде не написано — программа считает его сама. Скобки (chislo * i): посчитать, потом приклеить.

5Пять вопросов подряд (кульминация)

let chislo = 7for (let i = 1; i <= 5; i++) {  let otvet = Number(prompt(chislo + " × " + i + " = ?"))  if (otvet === chislo * i) {    console.log("✅ Верно!")  } else {    console.log("❌ Мимо. Правильно: " + (chislo * i))  }}
Строка let i = 3 ушла — теперь i заводит сам цикл. Скобки в скобках: if внутри for.

6Копилка очков

let chislo = 7let ochki = 0for (let i = 1; i <= 5; i++) {  let otvet = Number(prompt(chislo + " × " + i + " = ?"))  if (otvet === chislo * i) {    console.log("✅ Верно!")    ochki = ochki + 1  } else {    console.log("❌ Мимо. Правильно: " + (chislo * i))  }}console.log("Очки: " + ochki + " из 5")Очки: 4 из 5
Копилка: завести до цикла · растить внутри (в ветке «верно») · печатать после.

7Оценка в конце

if (ochki === 5) {  console.log("🏆 Идеально!")} else if (ochki >= 3) {  console.log("👍 Неплохо после каникул")} else {  console.log("Ещё разок — станет легче")}
