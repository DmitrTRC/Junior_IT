в ряду 5 бусин — сколько всего в узоре?
// открой F12 → Console и вставьlet ochki = 0for (let i = 1; i <= 3; i++) {  let otvet = Number(prompt("В узоре " + i + " рядов по 5 бусин — сколько всего?"))  if (otvet === 5 * i) {    ochki = ochki + 1    console.log("✦ Верно, ряд готов")  } else {    console.log("· Мимо. Было: " + (5 * i))  }}alert("🎨 Лиза, собрано рядов: " + ochki + " из 3")
