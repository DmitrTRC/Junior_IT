Лестница языков

Чем ниже — тем ближе к железу и быстрее. Чем выше — тем понятнее человеку.

// hello world

Один «Привет!» — два языка

section .data
  msg db "Привет!", 10
  len equ $ - msg
section .text
  global _start
_start:
  mov rax, 1        ; системный вызов write
  mov rdi, 1        ; куда: на экран
  mov rsi, msg      ; что: адрес текста
  mov rdx, len      ; сколько байт
  syscall
  mov rax, 60       ; системный вызов exit
  xor rdi, rdi
  syscall
=
console.log("Привет!")
